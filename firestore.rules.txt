// Règles de sécurité Firestore pour Récré
// ----------------------------------------
// À coller dans Firebase Console → Firestore Database → onglet "Règles",
// puis clique sur "Publier".
//
// Ce que ça fait :
//  - tout le monde peut LIRE les scores (nécessaire pour afficher les classements)
//  - n'importe qui peut AJOUTER un score, mais uniquement s'il a la bonne forme
//    (un des 5 jeux connus, un pseudo de 1 à 18 caractères, une valeur numérique
//    raisonnable)
//  - personne ne peut modifier ni supprimer un score déjà enregistré
//
// C'est un système sans compte, pensé pour un petit groupe d'amis de confiance :
// ça filtre les erreurs de forme, pas la triche délibérée.

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /scores/{scoreId} {
      allow read: if true;

      allow create: if request.resource.data.keys().hasOnly(['game', 'player', 'value', 'meta', 'date', 'createdAt'])
                    && request.resource.data.game is string
                    && request.resource.data.game in ['reflex', 'memory', 'taprush', 'daily', 'runner', 'mole', 'math', 'snake', 'aim', 'chrono', 'marches', 'decoupe', 'bille']
                    && request.resource.data.player is string
                    && request.resource.data.player.size() > 0
                    && request.resource.data.player.size() <= 18
                    && request.resource.data.value is number
                    && request.resource.data.value >= 0
                    && request.resource.data.value < 1000000;

      allow update, delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
